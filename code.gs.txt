function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    // Validate action
    var validActions = ['log', 'holiday', 'delete_holiday', 'sync_day', 'create_team', 'add_member'];
    if (validActions.indexOf(data.action) === -1) {
      return ContentService.createTextOutput(JSON.stringify({ status: "ignored" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var tz = ss.getSpreadsheetTimeZone(); 
    
    // Map the app's team names to your exact Sheet tab names
    var teamMap = {
      "Digi Yatra": "Digi Yatra",
      "OCR": "OCR", // Accounts for the trailing space in your sheet
      "FHIR": "FHIR",
      "MIRTH Connect": "MIRTH CONNECT",
      "ChatBot": "ChatBot",
      "Blood Connect": "Blood Connect"
    };
    
    // ==========================================
    // HELPER: DATE PARSING (Timezone Safe)
    // ==========================================
    var todayStr = data.date || ""; // "YYYY-MM-DD"
    var displayToday = "";
    var tomorrowStr = "";
    var displayTomorrow = "";
    
    if (todayStr) {
      var parts = todayStr.split('-'); 
      var y = parseInt(parts[0], 10);
      var m = parseInt(parts[1], 10);
      var d = parseInt(parts[2], 10);
      
      displayToday = String(d).padStart(2, '0') + "/" + String(m).padStart(2, '0') + "/" + y;

      // Calculate tomorrow perfectly without timezone shifting bugs
      var daysInMonth = new Date(y, m, 0).getDate();
      var tm_d = d + 1, tm_m = m, tm_y = y;
      if (tm_d > daysInMonth) { tm_d = 1; tm_m++; if (tm_m > 12) { tm_m = 1; tm_y++; } }
      tomorrowStr = tm_y + "-" + String(tm_m).padStart(2, '0') + "-" + String(tm_d).padStart(2, '0');
      displayTomorrow = String(tm_d).padStart(2, '0') + "/" + String(tm_m).padStart(2, '0') + "/" + tm_y;
    }
    
    // Robust function to parse dates from the sheet into YYYY-MM-DD
    function parseSheetDate(cellVal) {
      if (!cellVal) return "";
      if (cellVal instanceof Date) return Utilities.formatDate(cellVal, tz, "yyyy-MM-dd");
      var str = String(cellVal).trim();
      
      var ymdmatch = str.match(/(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);
      if (ymdmatch) return ymdmatch[1] + "-" + ymdmatch[2].padStart(2, '0') + "-" + ymdmatch[3].padStart(2, '0');
      
      var dmymatch = str.match(/(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})/);
      if (dmymatch) return dmymatch[3] + "-" + dmymatch[2].padStart(2, '0') + "-" + dmymatch[1].padStart(2, '0');
      
      return str;
    }

    // Helper to find header and target row
    function getRowIndexes(sheet, targetDateStr) {
      var values = sheet.getDataRange().getValues();
      var headerIdx = -1, targetIdx = -1;
      for (var i = 0; i < Math.min(20, values.length); i++) {
        if (values[i] && values[i][0] && String(values[i][0]).replace(/\s/g, '').toLowerCase().startsWith("date")) { 
          headerIdx = i; break; 
        }
      }
      if (headerIdx === -1) headerIdx = 6;
      for (var i = headerIdx + 1; i < values.length; i++) {
        if (parseSheetDate(values[i][0]) === targetDateStr) { targetIdx = i + 1; break; }
      }
      return { header: headerIdx, target: targetIdx, values: values };
    }

    // ==========================================
    // ACTION: CREATE TEAM
    // ==========================================
    if (data.action === 'create_team') {
      var sheetName = teamMap[data.team] || data.team;
      createTeamSheet(ss, sheetName);
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // ACTION: ADD MEMBER
    // ==========================================
    if (data.action === 'add_member') {
      var sheetName = teamMap[data.team] || data.team;
      var sheet = ss.getSheetByName(sheetName);
      if (!sheet) {
        sheet = createTeamSheet(ss, sheetName);
      }
      var memberName = data.name.trim();
      if (!memberName) {
        return ContentService.createTextOutput(JSON.stringify({ status: "ignored", reason: "empty name" })).setMimeType(ContentService.MimeType.JSON);
      }
      
      // 1. Add to top-level "Team Members" list in Column B
      var valuesColB = sheet.getRange("B3:B100").getValues();
      var memberExistsB = false;
      var firstEmptyRowB = -1;
      for (var i = 0; i < valuesColB.length; i++) {
        var val = String(valuesColB[i][0]).trim();
        if (val === memberName) {
          memberExistsB = true;
          break;
        }
        if (val === "" && firstEmptyRowB === -1) {
          firstEmptyRowB = i + 3;
        }
      }
      if (!memberExistsB) {
        if (firstEmptyRowB === -1) firstEmptyRowB = valuesColB.length + 3;
        sheet.getRange(firstEmptyRowB, 2).setValue(memberName);
        var maxBoxRow = Math.max(5, firstEmptyRowB);
        sheet.getRange("A2:C" + maxBoxRow).setBorder(true, true, true, true, true, true, "#7f8c8d", SpreadsheetApp.BorderStyle.SOLID);
      }
      
      // 2. Add to table headers in Row 7 (Column D+)
      var lastCol = sheet.getMaxColumns();
      var headers = sheet.getRange(7, 1, 1, lastCol).getValues()[0];
      var memberExistsCol = false;
      var firstEmptyCol = -1;
      for (var c = 3; c < headers.length; c++) {
        var hName = String(headers[c]).trim();
        if (hName === memberName) {
          memberExistsCol = true;
          break;
        }
        if (hName === "" && firstEmptyCol === -1) {
          firstEmptyCol = c + 1;
        }
      }
      if (!memberExistsCol) {
        if (firstEmptyCol === -1) {
          sheet.insertColumnAfter(lastCol);
          firstEmptyCol = lastCol + 1;
        }
        sheet.getRange(7, firstEmptyCol).setValue(memberName)
          .setBackground("#1b5e20")
          .setFontColor("#ffffff")
          .setFontWeight("bold")
          .setHorizontalAlignment("center");
        sheet.setColumnWidth(firstEmptyCol, 120);
        sheet.getRange(7, firstEmptyCol).setBorder(true, true, true, true, true, true, "#7f8c8d", SpreadsheetApp.BorderStyle.SOLID);
      }
      
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // ACTION: HOLIDAYS (CREATE & DELETE)
    // ==========================================
    if (data.action === 'holiday' || data.action === 'delete_holiday') {
      for (var teamKey in teamMap) {
        var sheet = ss.getSheetByName(teamMap[teamKey]);
        if (!sheet) continue;
        
        var loc = getRowIndexes(sheet, todayStr);
        
        if (data.action === 'holiday') {
          if (loc.target === -1) {
            sheet.appendRow([displayToday, data.holidayName, data.holidayName]);
          } else { 
            sheet.getRange(loc.target, 2).setValue(data.holidayName); 
            sheet.getRange(loc.target, 3).setValue(data.holidayName); 
          }
        } else if (data.action === 'delete_holiday' && loc.target !== -1) {
          // Wipe the holiday data
          sheet.getRange(loc.target, 2).clearContent();
          sheet.getRange(loc.target, 3).clearContent();
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // ACTION: SYNC DAY (THE CRUD OVERWRITE FUNCTION)
    // ==========================================
    if (data.action === 'sync_day') {
      var sheetName = teamMap[data.team] || data.team;
      var sheet = ss.getSheetByName(sheetName);
      if (!sheet) return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sheet not found" })).setMimeType(ContentService.MimeType.JSON);
      
      var loc = getRowIndexes(sheet, todayStr);
      
      if (loc.target === -1 && data.logs.length > 0) {
        sheet.appendRow([displayToday]);
        loc = getRowIndexes(sheet, todayStr); // Refresh to get the new row
      }
      
      if (loc.target !== -1) {
        // 1. Wipe the Logs (Col 3) and Hours (Col 4+) but LEAVE Goals (Col 2) intact!
        var maxCol = sheet.getMaxColumns();
        if (maxCol >= 3) {
          sheet.getRange(loc.target, 3, 1, maxCol - 2).clearContent();
        }
        
        // 2. Rebuild the row from scratch
        var combinedLog = "", combinedGoal = "";
        var headers = sheet.getDataRange().getValues()[loc.header];
        
        for (var i = 0; i < data.logs.length; i++) {
          var log = data.logs[i];
          if (log.todayLog) combinedLog += (combinedLog ? "\n\n" : "") + "• " + log.name + ": " + log.todayLog;
          // Ignore placeholders like "dont type anything"
          if (log.tomorrowGoal && String(log.tomorrowGoal).toLowerCase().indexOf("dont type") === -1) {
            combinedGoal += (combinedGoal ? "\n\n" : "") + "• " + log.name + ": " + log.tomorrowGoal;
          }
          
          if (log.hours && String(log.hours).trim() !== "") {
            var memberColIdx = -1;
            var searchName = log.name.trim().toLowerCase().split(" ")[0];
            for (var c = 3; c < headers.length; c++) {
              if (String(headers[c]).trim().toLowerCase().indexOf(searchName) !== -1) { memberColIdx = c + 1; break; }
            }
            if (memberColIdx === -1) {
              memberColIdx = headers.length < 3 ? 4 : headers.length + 1;
              sheet.getRange(loc.header + 1, memberColIdx).setValue(log.name);
              headers = sheet.getDataRange().getValues()[loc.header]; // update headers
            }
            sheet.getRange(loc.target, memberColIdx).setValue(log.hours + " hrs");
          }
        }
        
        // Write combined logs to today's row
        if (combinedLog) sheet.getRange(loc.target, 3).setValue(combinedLog);
        
        // Write combined goals to TOMORROW'S row (Fixed to look forward, not backwards)
        if (combinedGoal) {
          var tmwLoc = getRowIndexes(sheet, tomorrowStr);
          if (tmwLoc.target === -1) {
             sheet.appendRow([displayTomorrow]);
             tmwLoc = getRowIndexes(sheet, tomorrowStr);
          }
          sheet.getRange(tmwLoc.target, 2).setValue(combinedGoal);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // ACTION: STANDARD LOG APPEND (Restored)
    // ==========================================
    if (data.action === 'log') {
      var sheetName = teamMap[data.team] || data.team;
      var sheet = ss.getSheetByName(sheetName);
      
      if (!sheet) return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sheet tab not found: " + sheetName })).setMimeType(ContentService.MimeType.JSON);

      var todayLoc = getRowIndexes(sheet, todayStr);
      var tomorrowLoc = getRowIndexes(sheet, tomorrowStr);

      // Create rows if they are missing
      if (todayLoc.target === -1) {
        sheet.appendRow([displayToday]);
        todayLoc = getRowIndexes(sheet, todayStr);
      }
      if (tomorrowLoc.target === -1) {
        sheet.appendRow([displayTomorrow]);
        tomorrowLoc = getRowIndexes(sheet, tomorrowStr);
      }

      // Insert Task Completed (Today)
      if (data.todayLog && String(data.todayLog).trim() !== "") {
        var logCell = sheet.getRange(todayLoc.target, 3);
        var extLog = logCell.getValue();
        var newLog = "• " + data.name + ": " + data.todayLog;
        logCell.setValue(extLog ? extLog + "\n\n" + newLog : newLog);
      }

      // Insert Tomorrow's Goal
      if (data.tomorrowGoal && String(data.tomorrowGoal).trim() !== "") {
        var goalCell = sheet.getRange(tomorrowLoc.target, 2);
        var extGoal = goalCell.getValue();
        var newGoal = "• " + data.name + ": " + data.tomorrowGoal;
        var finalGoal = (extGoal && String(extGoal).toLowerCase().indexOf("dont type") === -1) ? extGoal + "\n\n" + newGoal : newGoal;
        goalCell.setValue(finalGoal);
      }

      // Insert Hours
      if (data.hours && String(data.hours).trim() !== "") {
        var headers = todayLoc.values[todayLoc.header];
        var memberColIdx = -1;
        var searchName = data.name.trim().toLowerCase().split(" ")[0]; 
        
        for (var c = 3; c < headers.length; c++) {
          var headName = String(headers[c]).trim().toLowerCase();
          if (headName && (headName.indexOf(searchName) !== -1 || searchName.indexOf(headName) !== -1)) {
            memberColIdx = c + 1; break;
          }
        }
        
        if (memberColIdx === -1) {
          memberColIdx = headers.length < 3 ? 4 : headers.length + 1;
          sheet.getRange(todayLoc.header + 1, memberColIdx).setValue(data.name);
        }
        
        var hrCell = sheet.getRange(todayLoc.target, memberColIdx);
        var extHrs = hrCell.getValue();
        var hrText = data.hours + " hrs";
        hrCell.setValue(extHrs ? extHrs + "\n" + hrText : hrText);
      }

      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function createTeamSheet(ss, teamName) {
  var sheet = ss.getSheetByName(teamName);
  if (sheet) return sheet;
  
  sheet = ss.insertSheet(teamName);
  sheet.setHideGridlines(false);
  
  // Set column widths
  sheet.setColumnWidth(1, 130); // Date
  sheet.setColumnWidth(2, 350); // Task Given
  sheet.setColumnWidth(3, 350); // Task Completed
  
  // Row 1: Title Banner
  sheet.getRange("A1:C1").merge().setValue(teamName + " Project Daily log")
    .setBackground("#27ae60")
    .setFontColor("#ffffff")
    .setFontSize(14)
    .setFontWeight("bold")
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle");
  sheet.setRowHeight(1, 40);
  
  // Row 2: Team Info Headers
  sheet.getRange("A2").setValue("Team")
    .setBackground("#f39c12")
    .setFontColor("#000000")
    .setFontWeight("bold")
    .setHorizontalAlignment("center");
    
  sheet.getRange("B2:C2").merge().setValue("Team Members")
    .setBackground("#f39c12")
    .setFontColor("#000000")
    .setFontWeight("bold")
    .setHorizontalAlignment("center");
  
  // Row 3: Write team name in A3
  sheet.getRange("A3").setValue(teamName)
    .setFontWeight("bold")
    .setHorizontalAlignment("center");
    
  // Default empty rows for box (A2:C5)
  sheet.getRange("A2:C5").setBorder(true, true, true, true, true, true, "#7f8c8d", SpreadsheetApp.BorderStyle.SOLID);
  
  // Row 7: Logs Table Headers
  sheet.getRange("A7").setValue("Date")
    .setBackground("#1b5e20")
    .setFontColor("#ffffff")
    .setFontWeight("bold")
    .setHorizontalAlignment("center");
    
  sheet.getRange("B7").setValue("Task Given (Prototype expected) / Goal for the day")
    .setBackground("#1b5e20")
    .setFontColor("#ffffff")
    .setFontWeight("bold");
    
  sheet.getRange("C7").setValue("Task completed (What was presented)")
    .setBackground("#1b5e20")
    .setFontColor("#ffffff")
    .setFontWeight("bold");
    
  sheet.getRange("A7:C7").setBorder(true, true, true, true, true, true, "#7f8c8d", SpreadsheetApp.BorderStyle.SOLID);
  
  return sheet;
}
